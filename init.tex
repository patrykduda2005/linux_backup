(make-variable-buffer-local
 (defvar zeszyt-mode nil
   "Toggle zeszyt-mode"))

(defvar zeszyt-mode (make-sparse-keymap)
  "Skroty do zeszytu")

(defun zeszyt-mode (&optional ARG)
  (interactive (list 'toggle))
  (setq zeszyt-mode
        (if (eq ARG 'toggle)
            (not zeszyt-mode)
          (> ARG 0)))
  (if zeszyt-mode
      (message "zeszyt mode odpalony")
    (message "zeszyt mode wypalony")))
